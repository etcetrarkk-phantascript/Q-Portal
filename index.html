/**
 * RUHUL KABIR EXAM PORTAL — GOOGLE SHEET BACKEND (Code.gs)
 * Stores Users (with Gmail 2FA & Admin Approval), Questions (Class/Subject/Paper/Chapter),
 * and Exam Attempts (1st/2nd/3rd attempts + PDF Type 2 Approval) inside this Google Sheet.
 */

function setupExamPortalSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = {
    'Users': ['Name', 'Roll', 'Class', 'Gmail', 'Password', 'Role', 'TwoFactorVerified', 'AdminApproved', 'CreatedAt'],
    'Questions': ['QSetId', 'Class', 'Subject', 'Paper', 'Chapter', 'DurationMins', 'PosMarks', 'NegMarks', 'QuestionsJSON'],
    'ExamAttempts': ['AttemptId', 'QSetId', 'Gmail', 'Name', 'Roll', 'Class', 'Subject', 'Paper', 'Chapter', 'AttemptNumber', 'DateStr', 'Correct', 'Wrong', 'Skipped', 'ObtainedMarks', 'TotalMarks', 'AnswersJSON', 'Pdf2Status'],
    'OtpCodes': ['Gmail', 'OtpCode', 'Purpose', 'ExpiresAt']
  };

  Object.keys(sheets).forEach(function(name) {
    var sh = ss.getSheetByName(name);
    if (!sh) {
      sh = ss.insertSheet(name);
      sh.appendRow(sheets[name]);
    }
  });
}

function jsonResponse_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  setupExamPortalSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var qRows = ss.getSheetByName('Questions').getDataRange().getValues().slice(1);
  var aRows = ss.getSheetByName('ExamAttempts').getDataRange().getValues().slice(1);
  return jsonResponse_({
    status: 'ok',
    questionsCount: qRows.length,
    attemptsCount: aRows.length
  });
}

function doPost(e) {
  setupExamPortalSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var body = JSON.parse(e.postData.contents || '{}');
  var action = body.action;

  // 1. Send 6-Digit 2FA OTP to valid @gmail.com address
  if (action === 'send2faOtp') {
    var email = String(body.email || '').trim().toLowerCase();
    if (!/^[a-z0-9._%+-]+@gmail.com$/.test(email)) {
      return jsonResponse_({ ok: false, error: 'Only valid @gmail.com accounts are allowed.' });
    }
    var otp = String(Math.floor(100000 + Math.random() * 900000));
    ss.getSheetByName('OtpCodes').appendRow([email, otp, body.purpose || 'register', new Date(Date.now() + 10 * 60000)]);
    MailApp.sendEmail({
      to: email,
      subject: 'Your 2FA Verification Code — Ruhul Kabir Exam Portal',
      body: 'Assalamu Alaikum,

Your 6-digit 2FA verification code is: ' + otp + '

Exam Taker: Ruhul Kabir'
    });
    return jsonResponse_({ ok: true, message: '2FA OTP sent to ' + email });
  }

  // 2. Save Question Set by Class -> Subject -> Paper -> Chapter
  if (action === 'saveQuestions') {
    var qSheet = ss.getSheetByName('Questions');
    qSheet.appendRow([
      body.id || ('qset_' + Date.now()),
      body.className,
      body.subject,
      body.paper,
      body.chapter,
      body.durationMins,
      body.posMarks,
      body.negMarks,
      JSON.stringify(body.questions)
    ]);
    return jsonResponse_({ ok: true });
  }

  // 3. Record Exam Attempt (tracks 1st, 2nd, 3rd attempt per user & chapter)
  if (action === 'recordAttempt') {
    var aSheet = ss.getSheetByName('ExamAttempts');
    aSheet.appendRow([
      body.attemptId,
      body.qsetId,
      body.userEmail,
      body.userName,
      body.userRoll,
      body.className,
      body.subject,
      body.paper,
      body.chapter,
      body.attemptNumber,
      body.dateStr,
      body.correct,
      body.wrong,
      body.skipped,
      body.obtainedMarks,
      body.totalMarks,
      JSON.stringify(body.answers),
      body.pdf2Status || 'none'
    ]);
    return jsonResponse_({ ok: true });
  }

  return jsonResponse_({ ok: true });
}
